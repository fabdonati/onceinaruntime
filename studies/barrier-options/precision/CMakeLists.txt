add_library(barrier_float_core SHARED precision.cpp)
target_include_directories(barrier_float_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(barrier_float_core PUBLIC barrier_core)
target_compile_options(barrier_float_core PUBLIC -fno-fast-math -ffp-contract=off)
set_target_properties(barrier_float_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

add_executable(barrier_precision_bench benchmark.cpp)
target_link_libraries(barrier_precision_bench PRIVATE barrier_float_core)
set_target_properties(barrier_precision_bench PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

    add_executable(barrier_precision_tests tests.cpp)
    target_link_libraries(barrier_precision_tests PRIVATE barrier_float_core)
    add_test(NAME barrier_precision_correctness COMMAND barrier_precision_tests)
endif()
